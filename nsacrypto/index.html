이 방법의 한계예요

암호가 뚫렸다는 얘기가 아니에요. 글도 새로운 공격이나 곡선의 약점이 아니라고 적었고, B-163 시드와 NIST 기저점 4개는 아직 설명하지 못했다고 해요. 제가 확인한 건 코드가 값을 다시 만든다는 것까지고, 삼십 분 만에 풀었다는 건 재지 않았어요. 이진 곡선 시각은 글이 가설로 적은 읽기입니다.

따라 하는 법, 파일 하나

  1. 하나. 아래 seeds.py 를 저장하고 python3 seeds.py 를 실행하세요. 시드 열네 줄이 1초도 안 걸려 나옵니다.
  2. 둘. basepoints.py 를 저장하고 python3 basepoints.py | grep -c True 를 실행하면 기저점 6개가 공개 값과 같은지 세어 줘요(6이 나오면 전부 일치).
  3. 셋. (선택) 표준 문서 SEC 2 PDF(secg.org/sec2-v2.pdf)를 텍스트로 뽑아 sec2_text.txt 로 같은 폴더에 두고 compare_sec2.py 를 실행하면 9개가 일치하는지 대조합니다.

from datetime import datetime, timezone

def unix(when):
    return int(datetime.fromisoformat(when).replace(tzinfo=timezone.utc).timestamp())

def candidate(t, n, per, shift, width):
    x, out = (t << 16) | 0x330E, ""                      # the state srand48(t) sets
    for i in range(per * n):
        x = (0x5DEECE66D * x + 0xB) % 2**48              # one step of lrand48
        if i >= per * (n - 1):
            out += "%0*X" % (width, x >> shift)
    return out

def base_point_x(seed, p, b):
    state = seed                         # zrstarts(seed)
    def limb():                          # zrandom(2^26): low 26 bits of the new state
        nonlocal state
        state = state * A % M
        return state % 2**26
    for _ in range(7):                   # seven single-limb draws come first
        limb()
    limbs = -(-p.bit_length() // 26) + 1 # zrandomb(p): limbs of p plus one, top limb first
    tries = 0
    while True:
        x = 0
        for _ in range(limbs):
            x = x << 26 | limb()
        x %= p
        tries += 1
        if pow(x**3 - 3*x + b, (p - 1) // 2, p) == 1:   # is x on the curve?
            return x, tries

for name, seed, p, b, gx in CURVES:
    x, tries = base_point_x(seed, p, b)
    print(f"{name:20} seed {seed}  tries {tries}  matches published Gx: {x == gx}")
compare_sec2.py (선택, 표준 문서와 대조)
# 코드 출력(seeds.py)의 시드를 SEC 2 문서 텍스트(sec2_text.txt)의 시드와 곡선별로 대조한다.
import re, subprocess, sys
sec = open("sec2_text.txt", encoding="utf8").read().split("\n")
names = {"secp192r1": "P-192", "secp224r1": "P-224", "secp256r1": "P-256", "secp384r1": "P-384", "secp521r1": "P-521",
         "sect233r1": "B-233", "sect283r1": "B-283", "sect409r1": "B-409", "sect571r1": "B-571"}
cur, pub = None, {}
for i, l in enumerate(sec):
    m = re.search(r"(secp\d+r1|sect\d+r1)", l)
    if m and "Recommended" in l: cur = m.group(1)
    m2 = re.search(r"S\s*=\s*((?:[0-9A-F]{8}\s*){5})", " ".join(sec[i:i + 4]).upper())
    if m2 and cur in names and names[cur] not in pub: pub[names[cur]] = m2.group(1).replace(" ", "")
mine = {}
for l in subprocess.run([sys.executable, "-I", "seeds.py"], capture_output=True, text=True).stdout.split("\n"):
    p = l.split()
    if len(p) >= 5 and re.fullmatch(r"[0-9A-F]{40}", p[-1]): mine[p[0]] = p[-1]
ok = 0
for k, v in pub.items():
    same = mine.get(k) == v; ok += same
    print(f"{k:6} {'일치' if same else '다름'}")
print(f"SEC 2 시드 {len(pub)}개 중 {ok}개 일치")
