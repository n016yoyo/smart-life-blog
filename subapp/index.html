화면 없는 터미널 가계부예요. 달력·휴대폰 앱·디자인·동기화는 없습니다. 기능 11개 목록과 검사 14개 항목은 제가 정했고, 구독 앱을 대신한다는 얘기가 아니에요. 이만큼은 이제 45초면 된다는 거예요. 제 환경과 버전 한 곳의 값이라 다르게 나올 수 있어요.

따라 하는 법, 하나 둘 셋

  1. 하나. 빈 폴더에 아래 「요청문」을 SPEC.md 로 저장하고, 그 폴더에서 time claude -p "$(cat SPEC.md)" --permission-mode acceptEdits --allowedTools "Bash,Write,Edit,Read" 를 실행하세요(제가 쓴 옵션 그대로, 걸린 시간은 time 이 알려 줘요).
  2. 둘. 아래 「검사 스크립트」를 check.js 로 저장하고 node check.js <budget.js 가 있는 폴더> 로 확인하세요. 마지막 줄이 RESULT 14/14 이면 통과입니다.
  3. 셋. 실행 전후에 claude -p "/usage" 를 한 번씩 돌려 주간 한도 숫자를 비교하세요.
요청문 (SPEC.md)
Build a command-line household budget app (가계부) as a single Node.js file `budget.js` in the current folder. No npm packages, Node built-ins only.
Data is stored as JSON in the file named by the environment variable BUDGET_FILE (default `data.json` in the current folder).
Commands (exact output formats matter, a test script will run them):

1. `node budget.js add <amount> <category> <note...> [--date YYYY-MM-DD]`  adds an expense (date defaults to today). Prints `added #<id>` (ids start at 1 and increase). Amount must be a positive integer, otherwise print an error to stderr and exit with code 1.
2. `node budget.js list [--month YYYY-MM]`  prints one line per expense: `#<id> <date> <category> <amount> <note>` in date order (oldest first). With --month only that month.
3. `node budget.js total --month YYYY-MM`  prints `total: <sum>`.
4. `node budget.js bycat --month YYYY-MM`  prints one line per category `<category>: <sum>` sorted by sum descending.
5. `node budget.js search <text>`  prints the list lines (same format as list) whose note contains the text (case-insensitive).
6. `node budget.js edit <id> --amount <n>`  changes the amount, prints `edited #<id>`.
7. `node budget.js delete <id>`  removes it, prints `deleted #<id>`. Unknown id: error to stderr, exit code 1.
8. `node budget.js budget set <category> <amount>`  sets a monthly limit for a category, prints `budget <category> <amount>`.
9. `node budget.js budget status --month YYYY-MM`  for each category that has a limit and whose month sum is above it prints `<category>: OVER by <difference>`; if none are over prints `all within budget`.
10. `node budget.js export csv <file>`  writes all expenses as CSV with header `id,date,category,amount,note`.
11. `node budget.js import csv <file>`  reads such a CSV and adds its rows as new expenses (new ids), prints `imported <count>`.
Work until the app is complete and try it yourself. Do not ask questions.
검사 스크립트 (check.js)
// 가계부 체크리스트 14개 , 스크립트가 판정한다(사람 아님). 사용: node check.js <budget.js 가 있는 폴더>
const { spawnSync } = require("child_process"), fs = require("fs"), path = require("path"), os = require("os");
const dir = path.resolve(process.argv[2] || "."), F = path.join(os.tmpdir(), "budget_check_" + process.pid + ".json");
const run = (...a) => spawnSync("node", ["budget.js", ...a], { cwd: dir, env: { ...process.env, BUDGET_FILE: F }, encoding: "utf8", timeout: 10000 });
const out = (...a) => (run(...a).stdout || "").trim();
const lines = (s) => s.split("\n").map((x) => x.trim()).filter(Boolean);
const today = new Date().toISOString().slice(0, 10);
try { fs.rmSync(F); } catch {}
const T = [];
const t = (name, fn) => { let ok = false, why = ""; try { ok = !!fn(); } catch (e) { why = String(e.message || e).slice(0, 80); } T.push({ name, ok, why }); };
t("1 지출 추가 → added #1", () => out("add", "5000", "food", "lunch", "--date", "2026-10-03") === "added #1");
t("2 id 증가(#2)", () => out("add", "12000", "food", "dinner", "with", "friends", "--date", "2026-10-05") === "added #2");
run("add", "30000", "transport", "taxi", "--date", "2026-10-07"); run("add", "9000", "food", "Coffee", "--date", "2026-11-02");
t("3 list 형식·날짜순", () => { const l = lines(out("list")); return l[0] === "#1 2026-10-03 food 5000 lunch" && l.length === 4 && l[3].startsWith("#4 2026-11-02"); });
t("4 list --month 필터", () => lines(out("list", "--month", "2026-10")).length === 3);
t("5 total", () => out("total", "--month", "2026-10") === "total: 47000");
t("6 bycat 내림차순", () => { const l = lines(out("bycat", "--month", "2026-10")); return l[0] === "transport: 30000" && l[1] === "food: 17000"; });
t("7 search 대소문자 무시", () => { const l = lines(out("search", "coffee")); return l.length === 1 && l[0].startsWith("#4 "); });
t("8 edit", () => out("edit", "1", "--amount", "6000") === "edited #1" && out("total", "--month", "2026-10") === "total: 48000");
t("9 delete", () => out("delete", "3") === "deleted #3" && out("total", "--month", "2026-10") === "total: 18000");
t("10 없는 id 삭제 → 종료코드 1", () => run("delete", "99").status === 1);
t("11 잘못된 금액 거부", () => run("add", "abc", "food", "x").status === 1 && run("add", "-5", "food", "x").status === 1);
t("12 예산 초과 표시", () => { out("budget", "set", "food", "10000"); return out("budget", "status", "--month", "2026-10") === "food: OVER by 8000"; });
t("13 CSV 내보내기", () => { const f = path.join(os.tmpdir(), "bx_" + process.pid + ".csv"); run("export", "csv", f); const l = lines(fs.readFileSync(f, "utf8")); return l[0] === "id,date,category,amount,note" && l.length >= 4; });
t("14 CSV 가져오기", () => { const f = path.join(os.tmpdir(), "bi_" + process.pid + ".csv"); fs.writeFileSync(f, "id,date,category,amount,note\n1,2026-12-01,etc,700,gum\n2,2026-12-02,etc,300,candy\n"); return out("import", "csv", f) === "imported 2" && lines(out("list", "--month", "2026-12")).length === 2; });
const pass = T.filter((x) => x.ok).length;
for (const x of T) console.log((x.ok ? "PASS " : "FAIL ") + x.name + (x.why ? "  (" + x.why + ")" : ""));
console.log(`RESULT ${pass}/${T.length}`);
try { fs.rmSync(F); } catch {}
process.exit(pass === T.length ? 0 : 1);
