TRAPS

먼저 알아야 할 함정 네 개

제가 만들면서 실제로 밟은 것들입니다. 이것만 알고 시작하면 한 번에 됩니다.

⚠️ 기존 글 수정 화면 주소는 postwrite?logNo=

구글에 나오는 PostUpdateForm이나 /postupdate/글번호는 404가 뜹니다. blog.naver.com/아이디/postwrite?logNo=글번호로 들어가야 기존 글이 로드된 에디터가 열립니다.

⚠️ 본문 전체 교체 금지 , 이미지가 다 날아갑니다

전체 선택 후 새 본문을 타이핑하면 글 속 사진이 전부 사라집니다. 부족한 섹션만 앵커 문단 뒤에 삽입하세요. 진단이 "이 단락이 없다"고 짚었으니, 없는 그 단락만 넣으면 됩니다.

⚠️ 수정 화면엔 임시저장이 없습니다

새 글 화면과 달리 기존 글 수정 화면은 발행을 눌러야만 반영됩니다. 그래서 자동화는 수정안 생성까지만 하고, 발행은 사람이 검토 후 누르는 구조로 만들었습니다. 실수로 글이 바뀔 일이 없습니다.

⚠️ 다음날 실행이 어제 증거를 덮어씁니다

수정안 파일명이 글번호로만 돼 있으면 매일 실행이 어제 파일을 덮습니다. 보관하려면 날짜 붙은 사본을 남기세요(fix_글번호_0820.md).

STEP 1

진단·수정안 만들기 , rank_fix.js

클로드 코드에 이렇게 말하면 아래 코드를 만들어 줍니다. 그대로 붙여넣어도 됩니다. (이전 가이드의 rank_watch.js가 만든 rank_history.json을 읽습니다.)

클로드에게: "rank_history에서 순위가 떨어진 키워드를 찾아서, 모바일 검색 결과에서 내 글이랑 상위 3개 글 본문을 모아서, 왜 밀렸는지랑 보강 본문을 클로드로 만들어서 파일로 저장하는 스크립트 만들어줘. 원본 백업도."
rank_fix.js (전문)
#!/usr/bin/env node
// 순위가 떨어진 내 글을 특정하고, 상위 글과 대조해 클로드가 수정안을 만든다.
// rank_watch.js(감시)의 다음 단계 , 2026-08-20 reel_rankfix 착수와 함께 제작.
//
// 하는 일(순서대로)
//   ①검색 결과에서 하락 키워드의 **내 글 실제 URL(logNo)** 과 상위 3개 글을 특정
//   ②내 글 전문 + 상위 글 제목·도입부를 수집(모바일 공개 페이지, 로그인 불필요)
//   ③원본을 evidence/ 에 백업(수정 전 상태 보존 , 실물 원칙)
//   ④claude CLI 로 "왜 밀렸나 + 어떻게 보강하나" 분석 → 수정안 생성
//   ⑤fix_<logNo>.md 저장 + 요약 출력
//
// ★발행은 하지 않는다. 에디터 반영(rank_apply.js)은 별도 단계 , 2026-07-19 사장님 결정
//   (자동은 임시저장까지, 발행 전 검토는 사람)을 기존 글 수정에도 그대로 따른다.
//
// 쓰는 법
//   node docs/naverblog-auto/rank_fix.js "재입고 알림"       # 키워드 지정
//   node docs/naverblog-auto/rank_fix.js                     # 이력에서 최근 하락 키워드 자동 선택
const fs = require("fs"), path = require("path");
const { execSync, execFileSync } = require("child_process");
const { chromium } = require(require.resolve("playwright", { paths: [path.resolve(__dirname, "../_engine")] }));

const D = __dirname;
const BLOG_ID = "n016yoyo";
const EV = path.resolve(D, "../reel_rankfix/evidence");
const HIST_FILE = path.join(D, "rank_history.json");
const CHR = execSync("ls /nix/store/*/bin/chromium | head -1").toString().trim();
const UA = "Mozilla/5.0 (iPhone; CPU iPhone OS 17_0 like Mac OS X) AppleWebKit/605.1.15 (KHTML, like Gecko) Version/17.0 Mobile/15E148 Safari/604.1";
const CLAUDE = "/home/user/.global_modules/bin/claude";
fs.mkdirSync(EV, { recursive: true });

const log = (...a) => console.log(new Date().toISOString().slice(11, 19), ...a);

// 하락 키워드 자동 선택: 이력의 마지막 두 날짜를 비교해 순위가 내려간 것 중 낙폭 최대
function pickDroppedKeyword() {
  const hist = JSON.parse(fs.readFileSync(HIST_FILE, "utf8"));
  const days = Object.keys(hist).sort();
  if (days.length < 2) throw new Error("이력이 2일치 미만");
  const [prev, cur] = [hist[days[days.length - 2]], hist[days[days.length - 1]]];
  let best = null;
  for (const kw of Object.keys(cur)) {
    const a = prev[kw] && prev[kw].rank, b = cur[kw] && cur[kw].rank;
    if (a && b && b > a && (!best || (b - a) > best.drop)) best = { kw, from: a, to: b, drop: b - a };
  }
  if (!best) throw new Error("하락한 키워드가 없다");
  return best;
}

(async () => {
  const kwArg = process.argv[2];
  const picked = kwArg ? { kw: kwArg } : pickDroppedKeyword();
  const KW = picked.kw;
  log(`대상 키워드: "${KW}"` + (picked.from ? ` (${picked.from}위 → ${picked.to}위)` : ""));

  const b = await chromium.launch({ headless: true, executablePath: CHR, args: ["--no-sandbox", "--disable-dev-shm-usage"] });
  const c = await b.newContext({ viewport: { width: 420, height: 900 }, locale: "ko-KR", userAgent: UA });
  const p = await c.newPage();

  // ① 검색 결과에서 내 글 + 상위 글 특정 (rank_watch 와 같은 카운트 규칙)
  await p.goto(`https://m.search.naver.com/search.naver?ssc=tab.m_blog.all&query=${encodeURIComponent(KW)}`, { waitUntil: "domcontentloaded", timeout: 40000 });
  await p.waitForTimeout(2200);
  const hrefs = await p.$$eval("a[href]", as => as.map(a => a.href || ""));
  const postOf = (href) => {
    try {
      const x = new URL(href, "https://m.search.naver.com");
      if (!/\d{4,}\/?$/.test(x.pathname)) return null;
      if (/(^|\.)blog\.naver\.com$/.test(x.hostname)) {
        const m = x.pathname.match(/^\/([A-Za-z0-9_-]+)\/(\d+)/);
        return m ? { key: "naver:" + m[1], owner: m[1], logNo: m[2], url: `https://m.blog.naver.com/${m[1]}/${m[2]}` } : null;
      }
      if (/naver\.com$/.test(x.hostname)) return null;
      return { key: "site:" + x.hostname, owner: x.hostname, logNo: null, url: href };
    } catch { return null; }
  };
  const posts = [];
  for (const h of hrefs) { const o = postOf(h); if (o && !posts.some(q => q.key === o.key)) posts.push(o); }
  const meIdx = posts.findIndex(q => q.key === "naver:" + BLOG_ID);
  if (meIdx < 0) throw new Error(`검색 결과에 내 글이 없다("${KW}")`);
  const me = posts[meIdx], tops = posts.slice(0, 3).filter(q => q.key !== me.key);
  log(`내 글: ${me.url} (현재 ${meIdx + 1}위 / 이 페이지 ${posts.length}개)`);

  // ② 본문 수집 , 내 글 전문 + 상위 글 제목·도입부
  const readPost = async (url, full) => {
    await p.goto(url, { waitUntil: "domcontentloaded", timeout: 40000 });
    await p.waitForTimeout(2000);
    return p.evaluate((full) => {
      const t = (document.querySelector(".se-title-text, .pcol1, h3.tit_h3") || {}).innerText || document.title;
      const bodyEl = document.querySelector(".se-main-container, #viewTypeSelector, .post_ct");
      let body = bodyEl ? bodyEl.innerText : "";
      body = body.replace(/\n{3,}/g, "\n\n").trim();
      return { title: t.trim(), body: full ? body : body.slice(0, 900) };
    }, full);
  };
  const mine = await readPost(me.url, true);
  log(`내 글 수집: "${mine.title}" (${mine.body.length}자)`);
  const rivals = [];
  for (const t of tops) {
    await p.waitForTimeout(1500);   // ★남의 서비스 , 간격은 안전장치
    try { const r = await readPost(t.url, false); rivals.push({ rank: posts.indexOf(t) + 1, ...r, url: t.url }); log(`상위 글 수집: ${posts.indexOf(t) + 1}위 "${r.title.slice(0, 40)}"`); }
    catch (e) { log(`  (상위 글 실패: ${e.message.slice(0, 40)})`); }
  }
  await b.close();

  // ③ 원본 백업 (수정 전 상태 , 실물 원칙: 사후 대조 가능해야 한다)
  const orig = path.join(EV, `original_${me.logNo}.md`);
  fs.writeFileSync(orig, `# 원본 백업 , ${new Date().toISOString()}\n키워드: ${KW} · 현재 ${meIdx + 1}위\nURL: ${me.url}\n\n## ${mine.title}\n\n${mine.body}\n`);
  log(`원본 백업: ${orig}`);

  // ④ 클로드 분석 , 자료를 파일로 만들어 넘긴다(화면=실행 명령 원칙: 이 파일이 곧 증거)
  const brief = path.join(EV, `analysis_input_${me.logNo}.md`);
  fs.writeFileSync(brief, [
    `# 네이버 블로그 순위 하락 분석 요청`,
    `검색어: "${KW}" , 내 글이 ${picked.from ? `${picked.from}위에서 ${picked.to}위로 떨어졌다` : `현재 ${meIdx + 1}위다`}.`,
    ``, `## 내 글 (전문)`, `제목: ${mine.title}`, ``, mine.body,
    ``, `## 지금 상위에 있는 글들 (제목 + 도입부)`,
    ...rivals.map(r => `\n### ${r.rank}위: ${r.title}\n${r.body}`),
  ].join("\n"));
  log("클로드 분석 중...");
  const prompt = `너는 네이버 블로그 SEO 편집자다. 첨부한 자료(내 글 전문 + 현재 상위 글들)를 읽고:
1) 내 글이 밀린 원인을 상위 글과의 실제 차이에서만 찾아라(추측 금지, 자료에 있는 것만).
2) 제목은 유지하고, 본문 보강안을 작성하라 , 기존 글의 사실·톤은 지키면서 부족한 부분(정보량·검색어 자연 포함·독자 질문 응답)을 채운 "완성 본문"으로.
3) 출력 형식: "## 진단"(3줄 이내) 다음 "## 보강 본문"(전문). 다른 말은 쓰지 마라.`;
  const out = execFileSync(CLAUDE, ["-p", prompt], { input: fs.readFileSync(brief), timeout: 300000, maxBuffer: 8 * 1024 * 1024 }).toString();

  // ⑤ 저장 + 요약
  const fix = path.join(EV, `fix_${me.logNo}.md`);
  fs.writeFileSync(fix, `# 수정안 , ${new Date().toISOString()}\n키워드: ${KW} · ${picked.from ? `${picked.from}→${picked.to}위` : `${meIdx + 1}위`} · logNo ${me.logNo}\n\n${out}\n`);
  log(`수정안 저장: ${fix}`);
  console.log("\n" + out.slice(0, 600) + "\n...");
  console.log(`\n다음 단계: 수정안 검토 후 rank_apply.js 로 에디터 반영(임시저장까지).`);
})().catch(e => { console.error("FAIL:", e.message.slice(0, 300)); process.exit(1); });

실행: node rank_fix.js "키워드" , 키워드를 생략하면 이력에서 가장 많이 떨어진 걸 자동으로 고릅니다.

STEP 2

에디터에 넣기 , rank_apply.js

수정안이 마음에 들면, 기존 글의 제자리에 새 섹션만 삽입합니다(이미지·기존 문단 보존). 함정 1·2·3이 전부 여기 반영돼 있습니다.

rank_apply.js (전문)
#!/usr/bin/env node
// rank_fix.js 의 보강안 중 **새 섹션만** 기존 글의 제자리에 삽입한다(삽입 모드).
// ★전체 교체는 하지 않는다 , 원글의 이미지·기존 문단을 전부 보존한다(2026-08-20 실측: 원글에 이미지 다수).
// ★임시저장까지만 한다. 발행 버튼은 누르지 않는다(2026-07-19 사장님 결정: 자동은 초안까지, 발행 전 검토는 사람).
//
// 쓰는 법
//   node docs/naverblog-auto/rank_apply.js <logNo> "<앵커 문구>" "<블록 시작 마커>" "<블록 끝 마커>"
//   예: node rank_apply.js 224316179238 "지금이 최저가인지" "재입고 알림 설정, 일단" "본론입니다"
const fs = require("fs"), path = require("path");
const { chromium } = require(require.resolve("playwright-core", { paths: [path.resolve(__dirname, "../_engine")] }));

const D = __dirname;
const BLOG_ID = "n016yoyo";
const EV = path.resolve(D, "../reel_rankfix/evidence");
const OUT = path.resolve(D, "../reel_rankfix/_work/apply");
const _R = path.resolve(__dirname, "../..");
const STATE = [_R + "/docs/scrape/creds/naver_state.json", _R + "/docs/scrape/.naver_state.json"].find(p => fs.existsSync(p));
const CHR = process.env.CHR || require("child_process").execSync("ls /nix/store/*/bin/chromium | head -1").toString().trim();
const log = (...a) => console.log(new Date().toISOString().slice(11, 19), ...a);
fs.mkdirSync(OUT, { recursive: true });

const [logNo, ANCHOR, MSTART, MEND] = process.argv.slice(2);
if (!logNo || !ANCHOR || !MSTART || !MEND) { console.error('사용: node rank_apply.js <logNo> "<앵커 문구>" "<블록 시작>" "<블록 끝>"'); process.exit(2); }

// 삽입 블록 추출: fix_<logNo>.md 에서 시작 마커 문단부터 끝 마커 문단까지
const raw = fs.readFileSync(path.join(EV, `fix_${logNo}.md`), "utf8");
const allParas = raw.split(/\n\s*\n/).map(s => s.replace(/\*\*/g, "").replace(/\n/g, " ").trim()).filter(Boolean);
const si = allParas.findIndex(t => t.includes(MSTART)), ei = allParas.findIndex(t => t.includes(MEND));
if (si < 0 || ei < 0 || ei < si) { console.error(`마커를 못 찾음: 시작 ${si} · 끝 ${ei}`); process.exit(2); }
const block = allParas.slice(si, ei + 1);
log(`삽입 블록: ${block.length}문단 · ${block.join("").length}자 ("${block[0].slice(0, 24)}…" ~ "…${block[block.length - 1].slice(-20)}")`);

(async () => {
  const b = await chromium.launch({ executablePath: CHR, headless: true, args: ["--no-sandbox", "--disable-gpu", "--disable-dev-shm-usage"] });
  const ctx = await b.newContext({
    storageState: STATE, viewport: { width: 1440, height: 900 },

STEP 3

아침마다 혼자 돌게 하기 , rank_daemon.sh

매일 아침 7시대에 순위를 재고, 떨어졌으면 수정안까지 만들어 텔레그램으로 보냅니다. 서버가 늘 켜져 있는 환경(개발 워크스페이스, 라즈베리파이, 집 서버 등)에서 돌립니다.

rank_daemon.sh (전문)
#!/usr/bin/env bash
# 순위 감시·수정안 데몬 , 하루 1회(KST 아침 7시대) rank_watch 를 돌리고,
# 하락이 감지되면 rank_fix(분석·수정안)까지 자동으로 만들어 텔레로 보고한다.
# ★발행은 하지 않는다(2026-08-20 사장님 확정: 자동화 범위 = 수정안 생성까지, 발행은 매번 승인).
# 기동: .idx/dev.nix onStart 가 자동 실행(worker_daemon 과 같은 패턴). 수동: bash rank_daemon.sh
cd /home/user/template || exit 1
PIDF=/tmp/rank_daemon.pid
LOG=/home/user/template/docs/naverblog-auto/rank_daemon.log
TG=/home/user/template/docs/telegram-claude-bridge/tg_send.sh

# 단일 인스턴스(멱등)
if [ -f "$PIDF" ] && kill -0 "$(cat "$PIDF" 2>/dev/null)" 2>/dev/null; then exit 0; fi
echo $$ > "$PIDF"
trap 'rm -f "$PIDF"' EXIT
echo "[$(date '+%F %T')] rank_daemon up (pid $$)" >> "$LOG"

while true; do
  # KST 시각·날짜 (서버는 UTC , date TZ 지정이 안 먹는 환경이라 파이썬 zoneinfo 로)
  read -r KH KD <<< "$(python3 -c "from datetime import datetime;from zoneinfo import ZoneInfo;n=datetime.now(ZoneInfo('Asia/Seoul'));print(n.hour, n.strftime('%Y-%m-%d'))")"
  MARK="/tmp/rank_daemon_done_$KD"
  if [ "$KH" -ge 7 ] && [ "$KH" -lt 9 ] && [ ! -f "$MARK" ]; then
    echo "[$(date '+%F %T')] 순위 측정 시작 (KST $KD ${KH}시)" >> "$LOG"
    node docs/naverblog-auto/rank_watch.js --notify >> "$LOG" 2>&1
    touch "$MARK"
    # 하락 감지 → rank_fix 자동(수정안 생성까지). pickDroppedKeyword 로직과 동일 기준.
    DROP=$(node -e '
      const h=require("/home/user/template/docs/naverblog-auto/rank_history.json");
      const days=Object.keys(h).sort(); if(days.length<2)process.exit(0);
      const [a,b]=[h[days[days.length-2]],h[days[days.length-1]]];
      let best=null;
      for(const kw of Object.keys(b)){const x=a[kw]&&a[kw].rank,y=b[kw]&&b[kw].rank;
        if(x&&y&&y>x&&(!best||y-x>best.d))best={kw,x,y,d:y-x};}
      if(best)console.log(`${best.kw}\t${best.x}\t${best.y}`);
    ' 2>>"$LOG")
    if [ -n "$DROP" ]; then
      KW=$(echo "$DROP" | cut -f1); FR=$(echo "$DROP" | cut -f2); TO=$(echo "$DROP" | cut -f3)
      echo "[$(date '+%F %T')] 하락 감지: $KW ${FR}→${TO}위 , rank_fix 실행" >> "$LOG"
      if node docs/naverblog-auto/rank_fix.js "$KW" >> "$LOG" 2>&1; then
        FIX=$(ls -t docs/reel_rankfix/evidence/fix_*.md 2>/dev/null | head -1)
        DIAG=$(sed -n '/^## 진단/,/^## 보강 본문/p' "$FIX" 2>/dev/null | sed '1d;$d' | head -4)
        bash "$TG" "📉 순위 하락 감지 → 수정안 준비 완료

키워드: $KW , ${FR}위 → ${TO}위

진단:
$DIAG

수정안: $FIX
★발행은 안 했습니다. 발행하려면 '발행해줘'라고 답 주세요(원본 백업 있음)." >> "$LOG" 2>&1
      else
        bash "$TG" "📉 순위 하락 감지($KW ${FR}→${TO}위)했는데 수정안 생성이 실패했습니다. rank_daemon.log 확인 필요." >> "$LOG" 2>&1
      fi
    fi
  fi
  sleep 600
done
실제 기록: 이 데몬이 첫 아침 실행에서 "재입고 알림 5위 → 7위" 하락을 감지해 수정안과 함께 텔레그램으로 보고했습니다. 사람이 한 일은 아침에 메시지를 읽은 것뿐입니다.
