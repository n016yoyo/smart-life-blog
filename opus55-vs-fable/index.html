돌린 명령

저장소 밖 빈 폴더에서, 설정·플러그인 빼고 모델만 바꿔서

#!/bin/bash
# 같은 요청 파일을 모델만 바꿔 claude -p 로 한 번씩. 설정·플러그인 제외(--setting-sources ""), 기본 effort.
# 사용: run.sh <폴더> <모델>
cd "$1"
S=$(date +%s.%N)
claude -p --model "$2" --setting-sources "" --tools "Read,Write,Edit" --permission-mode acceptEdits \
  --output-format stream-json --verbose "$(cat request.txt)" > stream.jsonl 2> stderr.txt
echo "wall $(python3 -c "print(round($(date +%s.%N)-$S,1))")" > wall.txt
